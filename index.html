import numpy as np

def calculate_leg_ik(x, y, z, L1=35.0, L2=95.0, L3=105.0):
    """Calculates 3-DOF joint angles (Coxa, Femur, Tibia) for a single leg in mm."""
    theta1 = np.arctan2(y, x)
    r = np.sqrt(x**2 + y**2) - L1
    d = np.sqrt(r**2 + z**2)
    
    # Law of Cosines for Femur and Tibia angles
    cos_theta3 = (d**2 - L2**2 - L3**2) / (2 * L2 * L3)
    theta3 = np.arccos(np.clip(cos_theta3, -1.0, 1.0))
    
    beta = np.arctan2(z, r)
    alpha = np.arccos((L2**2 + d**2 - L3**2) / (2 * L2 * d))
    theta2 = beta + alpha
    
    return np.degrees([theta1, theta2, theta3 - np.pi])
